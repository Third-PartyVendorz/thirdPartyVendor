List all docker containers, even not running:

docker ps -a

List docker networks:

docker network ls



Run from api subfolder:

docker build -t thirdparty:latest .


docker network create tpv-net


docker run -d \
  --name tpv-postgres \
  -e POSTGRES_DB=tpvdb \
  -e POSTGRES_USER=postgres \
  -e POSTGRES_PASSWORD=postgres \
  -p 5432:5432 \
  postgres:17


docker network connect tpv-net tpv-postgres


docker run -d \
  -p 8081:8081 \
  --network tpv-net \
  -e SPRING_DATASOURCE_URL=jdbc:postgresql://tpv-postgres:5432/tpvdb \
  -e SPRING_DATASOURCE_USERNAME=postgres \
  -e SPRING_DATASOURCE_PASSWORD=postgres \
  -e APP_JWT_SECRET=3deb18f27146742f1d6d7ccd218818bb1ee80ca7ce2b5737d70a391fdf226c54 \
  -e APP_JWT_EXPIRATION_MS=86400000 \
  --name tpv-api \
  thirdparty:latest

If not registered:

curl -X POST http://localhost:8081/auth/register \
  -H "Content-Type: application/json" \
  -d '{
    "firstName": "A",
    "lastName": "A",
    "phoneNumber": "555-0113",
    "dateOfBirth": "1990-05-15",
    "email": "ilas.P@example.com",
    "password": "SecurePassword123!"
  }'


  TOKEN=$(curl -s -X POST http://localhost:8081/auth/authenticate \
  -H "Content-Type: application/json" \
  -d '{
    "email": "ilas.P@example.com",
    "password": "SecurePassword123!"
  }' | jq -r '.jwtToken')

echo "JWT Token: $TOKEN"
